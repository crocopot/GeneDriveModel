""" Gene Drive Population Genetics Simulator ========================================= Models the spread of CRISPR-based gene drives through a diploid population. Alleles tracked: w = wild-type d = drive allele r = resistance allele (NHEJ-repaired, drive-resistant) Parameters: h : homing efficiency (probability drive copies itself in heterozygote) rho : resistance allele generation rate per homing attempt s : fitness cost of carrying the drive (applied to d-bearing genotypes) s_r : fitness cost of carrying the resistance allele (default 0; see below) d0 : initial drive allele frequency n : number of generations to simulate Fitness cost of resistance (s_r): In the original model, resistance alleles were assumed neutral (s_r = 0). In reality, NHEJ-generated resistance alleles are often indels that disrupt the target gene. If the target is fitness-relevant (e.g. doublesex, zpg), the resistance allele itself carries a reproductive burden. Even modest s_r > 0 slows resistance spread and extends drive persistence. Biological interpretation: s_r = 0.00 Resistance allele is fully neutral s_r = 0.01-0.05 Mild indel in non-essential region s_r = 0.05-0.15 Disruption of a fitness-relevant gene s_r = 0.15-0.30 Strong disruption; target is haploinsufficient s_r ~ 1.00 Lethal resistance allele Reference: Champer et al. (2020) PLOS Genetics — shows even small s_r values dramatically extend drive persistence against resistance. Usage: python gene_drive_simulator.py Produces: 1. Traditional vs Append-editing comparison 2. Parameter sweep heatmap 3. Custom single-drive simulation (interactive) 4. All three Citations: Crow & Kimura (1970) An Introduction to Population Genetics Theory. Deredec, Burt & Godfray (2008) Genetics 179(4):2013-2026. Champer et al. (2017) Nature Reviews Genetics 18:116-130. Unckless, Clark & Messer (2017) Genetics 205(2):827-841. Champer et al. (2020) PLOS Genetics. """ import numpy as np import matplotlib.pyplot as plt import matplotlib.gridspec as gridspec from dataclasses import dataclass from typing import Tuple # --------------------------------------------------------------------------- # Colour palette (colourblind-friendly) # --------------------------------------------------------------------------- BLUE = "#2171B5" GREEN = "#238B45" CORAL = "#D84B2A" AMBER = "#E08C1A" GRAY = "#636363" LIGHT = "#F7F7F7" plt.rcParams.update({ "figure.facecolor": LIGHT, "axes.facecolor": LIGHT, "axes.spines.top": False, "axes.spines.right": False, "axes.grid": True, "grid.color": "#DDDDDD", "grid.linewidth": 0.6, "font.family": "DejaVu Sans", "font.size": 11, }) # --------------------------------------------------------------------------- # Core data structure # --------------------------------------------------------------------------- @dataclass class DriveParams: """All parameters needed to define a gene drive scenario.""" name: str h: float # homing efficiency [0, 1] rho: float # resistance generation rate [0, 1] s: float # fitness cost of drive allele [0, 1] s_r: float = 0.0 # fitness cost of resistance allele [0, 1] d0: float = 0.01 # initial drive frequency [0, 1] n_gen: int = 60 # generations to simulate def summary(self) -> str: return ( f"{self.name}: h={self.h:.2f}, ρ={self.rho:.3f}, " f"s={self.s:.2f}, s_r={self.s_r:.2f}, d₀={self.d0:.3f}" ) # --------------------------------------------------------------------------- # Population model # --------------------------------------------------------------------------- def run_model(params: DriveParams) -> Tuple[np.ndarray, np.ndarray, np.ndarray]: """ Deterministic diploid allele-frequency recursion with three alleles: w, d, r. Each generation: 1. Hardy-Weinberg random mating -> genotype frequencies 2. Homing in w/d heterozygotes: h*(1-rho) fraction: w -> d (successful homing) h*rho fraction: w -> r (failed cut -> NHEJ resistance allele) (1-h) fraction: w stays w 3. Fitness selection: drive-bearing genotypes (wd, dd): fitness = 1 - s resistance-bearing genotypes (wr,dr,rr): fitness = 1 - s_r wild-type homozygotes (ww): fitness = 1.0 4. Renormalise allele frequencies. Returns ------- d_freq, w_freq, r_freq : np.ndarray of shape (n_gen+1,) """ h, rho, s, s_r, d0 = params.h, params.rho, params.s, params.s_r, params.d0 n = params.n_gen d = d0 w = 1.0 - d0 r = 0.0 d_arr = np.empty(n + 1) w_arr = np.empty(n + 1) r_arr = np.empty(n + 1) d_arr[0], w_arr[0], r_arr[0] = d, w, r fit_d = 1.0 - s fit_r = 1.0 - s_r for g in range(n): # --- Hardy-Weinberg genotype frequencies --- ww = w * w wd = 2 * w * d dd = d * d wr = 2 * w * r dr = 2 * d * r rr = r * r # --- Allele transmission after homing in wd heterozygotes --- nd = dd + wd * h * (1.0 - rho) # homed + dd nw = ww + wd * (1.0 - h) * (1.0 - rho) # unhomed wd + ww nr = rr + wr * 0.5 + dr * 0.5 + wd * rho # rr, mendelian r from wr/dr, NHEJ # --- Fitness selection --- total = nw * 1.0 + nd * fit_d + nr * fit_r if total <= 0: d_arr[g+1:] = d w_arr[g+1:] = w r_arr[g+1:] = r break d = (nd * fit_d) / total w = nw / total r = (nr * fit_r) / total # Re-normalise to remove floating-point drift _sum = d + w + r d /= _sum; w /= _sum; r /= _sum d_arr[g+1] = d w_arr[g+1] = w r_arr[g+1] = r return d_arr, w_arr, r_arr def invasion_generation(d_arr: np.ndarray, threshold: float = 0.5) -> int: """Return the first generation where drive frequency exceeds threshold, or -1.""" idxs = np.where(d_arr >= threshold)[0] return int(idxs[0]) if len(idxs) else -1 # --------------------------------------------------------------------------- # Figure 1: Single drive simulation # --------------------------------------------------------------------------- def plot_single(params: DriveParams, show: bool = True): d, w, r = run_model(params) gens = np.arange(len(d)) fig, ax = plt.subplots(figsize=(9, 5)) fig.patch.set_facecolor(LIGHT) ax.set_facecolor(LIGHT) ax.plot(gens, d * 100, color=BLUE, lw=2.5, label="Drive allele (d)") ax.plot(gens, w * 100, color=GREEN, lw=2.5, label="Wild-type (w)", linestyle="--") ax.plot(gens, r * 100, color=CORAL, lw=2.5, label="Resistance allele (r)", linestyle=":") inv = invasion_generation(d) if inv >= 0: ax.axvline(inv, color=BLUE, lw=1, linestyle="--", alpha=0.4) ax.text(inv + 0.5, 52, f"50% at gen {inv}", color=BLUE, fontsize=9) ax.set_xlim(0, params.n_gen) ax.set_ylim(0, 105) ax.set_xlabel("Generation", fontsize=12) ax.set_ylabel("Allele frequency (%)", fontsize=12) ax.set_title(f"Gene drive dynamics\n{params.summary()}", fontsize=11, pad=10) ax.legend(frameon=False, fontsize=10) final_d = d[-1] * 100 final_r = r[-1] * 100 textstr = ( f"Final drive: {final_d:.1f}%\n" f"Final resist: {final_r:.1f}%\n" f"50% invasion: {'gen ' + str(inv) if inv >= 0 else 'not reached'}" ) ax.text(0.98, 0.97, textstr, transform=ax.transAxes, fontsize=9, va='top', ha='right', bbox=dict(boxstyle='round,pad=0.4', facecolor='white', alpha=0.8, edgecolor='#CCCCCC')) if show: plt.tight_layout() plt.savefig("gene_drive_single.png", dpi=150, bbox_inches='tight') print("Saved: gene_drive_single.png") plt.show() # --------------------------------------------------------------------------- # Figure 2: Traditional vs Append-Editing comparison # --------------------------------------------------------------------------- def plot_comparison(show: bool = True): """ Compare traditional SpCas9 NGG-PAM drive vs append-editing DarT2-nSpCas9-NG. s_r = 0.10 for both: both drives target a fitness-relevant gene (e.g. doublesex or zpg), so NHEJ-generated resistance alleles that disrupt that gene carry a real reproductive cost. Sources: Hammond et al. 2016 (Nature Biotech) Kyrou et al. 2018 (Nature Biotech) Champer et al. 2020 (PLOS Genetics) """ traditional = DriveParams( name="Traditional CRISPR (SpCas9-NGG)", h=0.95, rho=0.08, s=0.05, s_r=0.10, d0=0.01, n_gen=60, ) append_edit = DriveParams( name="Append-editing (DarT2-nSpCas9-NG)", h=0.70, rho=0.015, s=0.03, s_r=0.10, d0=0.01, n_gen=60, ) td, tw, tr = run_model(traditional) ad, aw, ar = run_model(append_edit) gens = np.arange(len(td)) fig = plt.figure(figsize=(13, 9)) fig.patch.set_facecolor(LIGHT) gs = gridspec.GridSpec(2, 2, figure=fig, hspace=0.42, wspace=0.35) # Drive frequency ax1 = fig.add_subplot(gs[0, 0]) ax1.plot(gens, td * 100, color=BLUE, lw=2.5, label="Traditional") ax1.plot(gens, ad * 100, color=GREEN, lw=2.5, label="Append-editing", linestyle="--") ax1.set_title("Drive allele (d) frequency", fontsize=11) ax1.set_ylabel("Frequency (%)") ax1.set_ylim(0, 105) ax1.legend(frameon=False, fontsize=9) # Resistance frequency ax2 = fig.add_subplot(gs[0, 1]) ax2.plot(gens, tr * 100, color=CORAL, lw=2.5, label="Traditional") ax2.plot(gens, ar * 100, color=AMBER, lw=2.5, label="Append-editing", linestyle="--") ax2.set_title("Resistance allele (r) frequency", fontsize=11) ax2.set_ylabel("Frequency (%)") ax2.set_ylim(0, 105) ax2.legend(frameon=False, fontsize=9) # Wild-type frequency ax3 = fig.add_subplot(gs[1, 0]) ax3.plot(gens, tw * 100, color=BLUE, lw=2.5, label="Traditional", linestyle="--") ax3.plot(gens, aw * 100, color=GREEN, lw=2.5, label="Append-editing") ax3.set_title("Wild-type (w) frequency", fontsize=11) ax3.set_xlabel("Generation") ax3.set_ylabel("Frequency (%)") ax3.set_ylim(0, 105) ax3.legend(frameon=False, fontsize=9) # Summary bar chart ax4 = fig.add_subplot(gs[1, 1]) metrics = { "Final drive (%)": [td[-1]*100, ad[-1]*100], "Final resistance (%)": [tr[-1]*100, ar[-1]*100], "Peak drive (%)": [max(td)*100, max(ad)*100], } x = np.arange(len(metrics)) width = 0.32 bars_t = ax4.bar(x - width/2, [v[0] for v in metrics.values()], width, label="Traditional", color=BLUE, alpha=0.85) bars_a = ax4.bar(x + width/2, [v[1] for v in metrics.values()], width, label="Append-edit", color=GREEN, alpha=0.85) ax4.set_xticks(x) ax4.set_xticklabels(list(metrics.keys()), fontsize=9) ax4.set_ylabel("Frequency (%)") ax4.set_title("Summary statistics at gen 60", fontsize=11) ax4.legend(frameon=False, fontsize=9) ax4.set_ylim(0, 115) for bar in list(bars_t) + list(bars_a): h_val = bar.get_height() ax4.annotate(f"{h_val:.0f}", xy=(bar.get_x() + bar.get_width() / 2, h_val), xytext=(0, 3), textcoords="offset points", ha='center', fontsize=8) for ax in [ax1, ax2, ax3]: ax.set_xlabel("Generation") ax.set_xlim(0, 60) inv_t = invasion_generation(td) inv_a = invasion_generation(ad) param_text = ( f"Traditional: h={traditional.h}, ρ={traditional.rho}, " f"s={traditional.s}, s_r={traditional.s_r} " f"-> 50% invasion: {'gen ' + str(inv_t) if inv_t>=0 else 'not reached'}\n" f"Append-edit: h={append_edit.h}, ρ={append_edit.rho}, " f"s={append_edit.s}, s_r={append_edit.s_r} " f"-> 50% invasion: {'gen ' + str(inv_a) if inv_a>=0 else 'not reached'}" ) fig.text(0.5, 0.97, "Traditional CRISPR vs Append-Editing Gene Drive", ha='center', fontsize=14, fontweight='bold') fig.text(0.5, 0.935, param_text, ha='center', fontsize=9, color=GRAY, family='monospace') if show: plt.savefig("gene_drive_comparison.png", dpi=150, bbox_inches='tight') print("Saved: gene_drive_comparison.png") plt.show() # --------------------------------------------------------------------------- # Figure 3: Parameter sweep (h vs rho, at fixed s and s_r) # --------------------------------------------------------------------------- def plot_parameter_sweep(s_r: float = 0.10, show: bool = True): """ 2-D sweep over homing efficiency (h) and resistance rate (rho), showing final drive frequency as a heatmap, at fixed s and s_r. """ h_vals = np.linspace(0.50, 1.00, 30) rho_vals = np.linspace(0.00, 0.20, 30) final_d = np.zeros((len(rho_vals), len(h_vals))) base = DriveParams(name="sweep", h=0.0, rho=0.0, s=0.05, s_r=s_r, d0=0.01, n_gen=60) for i, rho in enumerate(rho_vals): for j, h in enumerate(h_vals): base.h = h base.rho = rho d_arr, _, _ = run_model(base) final_d[i, j] = d_arr[-1] * 100 fig, ax = plt.subplots(figsize=(8, 6)) fig.patch.set_facecolor(LIGHT) ax.set_facecolor(LIGHT) im = ax.contourf(h_vals, rho_vals, final_d, levels=20, cmap="RdYlGn") ax.contour(h_vals, rho_vals, final_d, levels=[50], colors="black", linewidths=1.5, linestyles="--") ax.text(0.97, 0.03, "-- 50% drive frequency contour", transform=ax.transAxes, fontsize=9, ha="right", bbox=dict(facecolor='white', alpha=0.7, edgecolor='none')) ax.plot(0.95, 0.08, 'o', ms=10, color=BLUE, label="Traditional CRISPR", zorder=5) ax.plot(0.70, 0.015, 's', ms=10, color=GREEN, label="Append-editing", zorder=5) ax.legend(frameon=True, fontsize=9, loc="upper right", framealpha=0.85, edgecolor="#CCCCCC") cbar = fig.colorbar(im, ax=ax, pad=0.02) cbar.set_label("Final drive frequency (%)", fontsize=11) ax.set_xlabel("Homing efficiency h", fontsize=12) ax.set_ylabel("Resistance generation rate rho", fontsize=12) ax.set_title( f"Drive outcome at generation 60\n(s=0.05, s_r={s_r:.2f}, d0=0.01)", fontsize=12, pad=10 ) if show: plt.tight_layout() plt.savefig("gene_drive_sweep.png", dpi=150, bbox_inches='tight') print("Saved: gene_drive_sweep.png") plt.show() # --------------------------------------------------------------------------- # Interactive CLI runner # --------------------------------------------------------------------------- def get_float(prompt: str, default: float, lo: float = 0.0, hi: float = 1.0) -> float: while True: raw = input(f" {prompt} [{default}]: ").strip() if raw == "": return default try: val = float(raw) if lo <= val <= hi: return val print(f" Please enter a value between {lo} and {hi}.") except ValueError: print(" Invalid input, please enter a number.") def get_int(prompt: str, default: int, lo: int = 1, hi: int = 500) -> int: while True: raw = input(f" {prompt} [{default}]: ").strip() if raw == "": return default try: val = int(raw) if lo <= val <= hi: return val print(f" Please enter an integer between {lo} and {hi}.") except ValueError: print(" Invalid input.") def interactive_single(): print("\n=== Custom single-drive simulation ===") print("Press Enter to accept the default value shown in brackets.\n") params = DriveParams( name = "Custom drive", h = get_float("Homing efficiency h", 0.95), rho = get_float("Resistance generation rate rho", 0.05, 0.0, 0.5), s = get_float("Fitness cost of drive s", 0.05), s_r = get_float("Fitness cost of resistance allele s_r", 0.00), d0 = get_float("Initial drive frequency d0", 0.01), n_gen = get_int( "Generations to simulate", 60, 1, 500), ) plot_single(params) #
